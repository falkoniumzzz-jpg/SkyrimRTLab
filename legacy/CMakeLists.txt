cmake_minimum_required(VERSION 3.24)
project(SkyrimRTLab LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "DXR requires Windows. Use a Windows cross toolchain for the probe or GitHub Actions for MSVC/plugins.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT RTLAB_DIRECTX_HEADERS)
  include(FetchContent)
  FetchContent_Declare(dxheaders GIT_REPOSITORY https://github.com/microsoft/DirectX-Headers.git
    GIT_TAG d873b344dc540898868697245f100c2a67fe68d9)
  set(DXHEADERS_BUILD_TEST OFF CACHE BOOL "" FORCE)
  set(DXHEADERS_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(dxheaders)
  set(RTLAB_DIRECTX_HEADERS "${dxheaders_SOURCE_DIR}/include")
endif()
find_program(DXC_EXECUTABLE dxc REQUIRED)
add_executable(RTLabProbe probe/main.cpp)
target_include_directories(RTLabProbe PRIVATE "${RTLAB_DIRECTX_HEADERS}")
target_link_libraries(RTLabProbe PRIVATE d3d12 d3d11 dxgi dxguid)
if(MSVC)
  target_compile_options(RTLabProbe PRIVATE /W4 /EHsc /permissive- /utf-8)
else()
  target_compile_definitions(RTLabProbe PRIVATE __REQUIRED_RPCNDR_H_VERSION__=475)
  target_compile_options(RTLabProbe PRIVATE -Wall -Wextra)
  target_link_options(RTLabProbe PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Probe.cso"
  COMMAND "${DXC_EXECUTABLE}" -T cs_6_5 -E Main -WX -Fo "${CMAKE_CURRENT_BINARY_DIR}/Probe.cso" "${CMAKE_CURRENT_SOURCE_DIR}/probe/Probe.hlsl"
  DEPENDS probe/Probe.hlsl VERBATIM)
add_custom_target(probe_shader ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/Probe.cso")
add_dependencies(RTLabProbe probe_shader)
install(TARGETS RTLabProbe RUNTIME DESTINATION .)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/Probe.cso" probe/Run-Probe.cmd DESTINATION .)

cmake_minimum_required(VERSION 3.24)
project(RTLabBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(rtlab-contract tests/contract.cpp)
target_include_directories(rtlab-contract PRIVATE include)
add_test(NAME contract COMMAND rtlab-contract)
if(WIN32)
    add_library(RTLabDiagnostics SHARED module/Diagnostics.cpp)
    target_include_directories(RTLabDiagnostics PRIVATE include)
    target_link_libraries(RTLabDiagnostics PRIVATE d3d11 d3d12 dxgi)
    set_target_properties(RTLabDiagnostics PROPERTIES OUTPUT_NAME "RTLab.Diagnostics")
    if(MSVC)
        target_compile_options(RTLabDiagnostics PRIVATE /W4 /WX /permissive-)
    endif()
endif()
if(WIN32)
    add_executable(rtlab-module-test tests/windows_module.cpp)
    target_include_directories(rtlab-module-test PRIVATE include)
    target_link_libraries(rtlab-module-test PRIVATE d3d11)
    add_test(NAME module-warp COMMAND rtlab-module-test $<TARGET_FILE:RTLabDiagnostics>)
endif()
